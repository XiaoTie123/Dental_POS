Add Nginx
=========
sudo vi /etc/nginx/sites-available/onedentalclinic.xyz

server {
    server_name onedentalclinic.xyz;
    index index.html index.htm;
    access_log /var/log/nginx/onedentalclinic.log;
    error_log  /var/log/nginx/onedentalclinic-error.log error;

    location / {
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header Host $http_host;
        proxy_pass http://127.0.0.1:8080;
        proxy_redirect off;
    }
}

sudo ln -s /etc/nginx/sites-available/onedentalclinic.xyz /etc/nginx/sites-enabled/onedentalclinic.xyz

sudo nginx -t

sudo nginx -s reload

sh startup.sh
